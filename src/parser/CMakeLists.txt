# T-SQL parsers: SqlScriptDOM's TSql<ver> grammars converted to ANTLR 4 by tools/g2to4 at build
# time (into ${CMAKE_CURRENT_BINARY_DIR}/generated), plus the hand-written parser bases.
set(TSQL_PARSER_ROOT ${CMAKE_CURRENT_SOURCE_DIR}/../..)

# Which grammars to build: any of the SqlScriptDOM parsers below (tsql::parse throws
# std::invalid_argument for the others). Each full parser takes about a minute to compile, and each
# of its rule translation units needs up to ~1.4 GB.
set(TSQL_PARSER_ALL_GRAMMARS TSql130 TSql140 TSql150 TSql160 TSql170 TSql180 TSqlFabricDW)
set(TSQL_PARSER_GRAMMARS "${TSQL_PARSER_ALL_GRAMMARS}" CACHE STRING
    "Semicolon-separated SqlScriptDOM grammars to build (subset of: ${TSQL_PARSER_ALL_GRAMMARS})")
if(NOT TSQL_PARSER_GRAMMARS)
    message(FATAL_ERROR "TSQL_PARSER_GRAMMARS is empty")
endif()
set(TSQL_PARSER_PARTS 5 CACHE STRING
    "Translation units each generated parser's rules are split into (plus one for its ATN data)")

# Compiling many parser translation units at once can exhaust memory, so with Ninja they share a
# job pool sized to half the memory free at configure time (1.6 GB per job). Other generators
# follow their own -j.
set(TSQL_PARSER_COMPILE_JOBS "" CACHE STRING
    "Concurrent tsql_parser compiles with Ninja (empty: half the free memory at configure time / 1.6 GB)")
set(parser_compile_jobs ${TSQL_PARSER_COMPILE_JOBS})
if(NOT parser_compile_jobs)
    cmake_host_system_information(RESULT free_mib QUERY AVAILABLE_PHYSICAL_MEMORY)
    math(EXPR parser_compile_jobs "${free_mib} / 2 / 1600")
    if(parser_compile_jobs LESS 1)
        set(parser_compile_jobs 1)
    endif()
endif()
set_property(GLOBAL APPEND PROPERTY JOB_POOLS tsql_parser_compile=${parser_compile_jobs})
message(STATUS "tsql_parser compile jobs: ${parser_compile_jobs}")

# ---- code generation (each step rewrites a file only when its content changes, so editing an
#      input re-runs the steps that read it but recompiles only what actually changed)
set(gen ${CMAKE_CURRENT_BINARY_DIR}/generated)
set(g2to4 ${TSQL_PARSER_ROOT}/tools/g2to4)
set(g2to4_py ${g2to4}/convert.py ${g2to4}/antlr2.py ${g2to4}/cs2cpp.py ${g2to4}/lexer.py ${g2to4}/model.py)
set(ssd_tsql ${TSQL_SSD_DIR}/SqlScriptDom/Parser/TSql)
# Override files are globbed with CONFIGURE_DEPENDS: adding or removing one re-runs CMake at the
# next build, and editing one re-runs the steps that read it.
file(GLOB_RECURSE common_overrides CONFIGURE_DEPENDS ${g2to4}/overrides/common/*)
# 1. grammar: <G>.g -> ANTLR 4 grammars, for every grammar (the shared support code is the union
#    over all of them, whichever are built)
set(support_inputs)
foreach(g IN LISTS TSQL_PARSER_ALL_GRAMMARS)
    set(g_overrides)
    if(IS_DIRECTORY ${g2to4}/overrides/${g})
        file(GLOB_RECURSE g_overrides CONFIGURE_DEPENDS ${g2to4}/overrides/${g}/*)
    endif()
    set(gdir ${gen}/${g}/grammar)
    add_custom_command(
        OUTPUT ${gdir}/${g}Lexer.g4 ${gdir}/${g}Parser.g4 ${gdir}/TSqlTokens.tokens ${gdir}/${g}.support.json
            ${gen}/reports/${g}.json
        COMMAND ${Python3_EXECUTABLE} -B ${g2to4}/convert.py grammar --strict --grammar ${g}
            --ssd ${TSQL_SSD_DIR} --ast-enums ${TSQL_AST_ENUMS_HPP} --out ${gdir} --report-dir ${gen}/reports
        DEPENDS ${g2to4_py} ${common_overrides} ${g_overrides} ${TSQL_AST_ENUMS_HPP}
            ${ssd_tsql}/${g}.g ${ssd_tsql}/TSqlTokenTypes.g ${ssd_tsql}/Ast.xml
        COMMENT "g2to4: converting ${g}.g to ANTLR 4"
        VERBATIM)
    list(APPEND support_inputs ${gdir}/${g}.support.json)
endforeach()
# 2. support: code shared by every grammar (also reads the hand-written sources for the
#    OptionsHelper classes they use)
file(GLOB parser_hand_sources CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/*.cpp ${CMAKE_CURRENT_SOURCE_DIR}/*.h
    ${CMAKE_CURRENT_SOURCE_DIR}/helpers/*)
set(support_headers)
foreach(f CodeGenerationSupporter.h TSqlParserResource.h Keywords.h OptionHelpers.h ParserEnums.h)
    list(APPEND support_headers ${gen}/support/${f})
endforeach()
add_custom_command(
    OUTPUT ${support_headers} ${gen}/reports/support.json
    COMMAND ${Python3_EXECUTABLE} -B ${g2to4}/convert.py support --strict --ssd ${TSQL_SSD_DIR}
        --ast-enums ${TSQL_AST_ENUMS_HPP} --out ${gen}/support --report ${gen}/reports/support.json ${support_inputs}
    DEPENDS ${g2to4_py} ${g2to4}/support.py ${support_inputs} ${parser_hand_sources} ${TSQL_AST_ENUMS_HPP}
    COMMENT "g2to4: generating the parser support code"
    VERBATIM)
# 3. antlr: ANTLR 4 tool -> C++, for the grammars being built
set(TSQL_PARSER_GRAMMAR_SOURCES)
set(TSQL_PARSER_GRAMMAR_HEADERS)
set(TSQL_PARSER_GRAMMARS_INC "// GENERATED by src/parser/CMakeLists.txt: TSQL_GRAMMAR(<grammar>, <SqlVersion>)\n")
foreach(g IN LISTS TSQL_PARSER_GRAMMARS)
    if(NOT g IN_LIST TSQL_PARSER_ALL_GRAMMARS)
        message(FATAL_ERROR "TSQL_PARSER_GRAMMARS: unknown grammar '${g}' (known: ${TSQL_PARSER_ALL_GRAMMARS})")
    endif()
    set(gdir ${gen}/${g}/grammar)
    set(odir ${gen}/${g})
    set(g_sources ${odir}/${g}Lexer.cpp ${odir}/${g}Parser.cpp)
    foreach(k RANGE 1 ${TSQL_PARSER_PARTS})
        list(APPEND g_sources ${odir}/${g}Parser_${k}.cpp)
    endforeach()
    set(g_headers ${odir}/${g}Lexer.h ${odir}/${g}Parser.h)
    add_custom_command(
        OUTPUT ${g_sources} ${g_headers}
        COMMAND ${Python3_EXECUTABLE} -B ${g2to4}/convert.py antlr --strict --grammar ${g} --grammar-dir ${gdir}
            --out ${odir} --java ${Java_JAVA_EXECUTABLE} --antlr-jar ${TSQL_ANTLR_JAR} --parts ${TSQL_PARSER_PARTS}
            --report ${gen}/reports/${g}.antlr.json
        BYPRODUCTS ${gen}/reports/${g}.antlr.json
        DEPENDS ${g2to4}/convert.py ${TSQL_ANTLR_JAR} ${gdir}/${g}Lexer.g4 ${gdir}/${g}Parser.g4 ${gdir}/TSqlTokens.tokens
        COMMENT "g2to4: running the ANTLR ${TSQL_ANTLR_VERSION} tool on ${g}"
        VERBATIM)
    string(REPLACE "TSql" "Sql" v ${g})            # TSql160 -> Sql160, TSqlFabricDW -> SqlFabricDW
    set(G ${g})
    set(V ${v})
    configure_file(ParseGrammar.cpp.in ${CMAKE_CURRENT_BINARY_DIR}/Parse${g}.cpp @ONLY)
    list(APPEND TSQL_PARSER_GRAMMAR_SOURCES ${g_sources} ${CMAKE_CURRENT_BINARY_DIR}/Parse${g}.cpp)
    list(APPEND TSQL_PARSER_GRAMMAR_HEADERS ${g_headers})
    # ANTLR output + converted actions: unused rule-context bindings and labels are expected
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        set_source_files_properties(${g_sources} PROPERTIES
            COMPILE_OPTIONS "-Wno-unused-variable;-Wno-unused-but-set-variable")
    endif()
    string(APPEND TSQL_PARSER_GRAMMARS_INC "TSQL_GRAMMAR(${g}, ${v})\n")
endforeach()
file(CONFIGURE OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/tsql_parser_grammars.inc CONTENT "${TSQL_PARSER_GRAMMARS_INC}")
message(STATUS "tsql_parser grammars: ${TSQL_PARSER_GRAMMARS}")
add_custom_target(tsql_parser_generate DEPENDS ${support_headers} ${TSQL_PARSER_GRAMMAR_SOURCES}
    ${TSQL_PARSER_GRAMMAR_HEADERS})
add_dependencies(tsql_parser_generate tsql_ast_generate)

add_library(tsql_parser STATIC
    ${TSQL_PARSER_GRAMMAR_SOURCES}
    ${TSQL_PARSER_GRAMMAR_HEADERS}
    ${support_headers}
    CsCompat.cpp
    ParserErrors.cpp
    TSqlLexerBase.cpp
    ParserRuntime.cpp
    TSql80ParserBase.cpp
    TSql90ParserBase.cpp
    TSql100ParserBase.cpp
    TSql110ParserBase.cpp
    TSql120ParserBase.cpp
    TSql130ParserBase.cpp
    TSql140ParserBase.cpp
    TSql150ParserBase.cpp
    TSql160ParserBase.cpp
    TSql170ParserBase.cpp
    TSql180ParserBase.cpp
    TSqlFabricDWParserBase.cpp
    VersioningVisitor.cpp
    Parse.cpp
)
add_dependencies(tsql_parser tsql_parser_generate)
set_property(TARGET tsql_parser PROPERTY JOB_POOL_COMPILE tsql_parser_compile)
add_library(tsql::parser ALIAS tsql_parser)
set_target_properties(tsql_parser PROPERTIES EXPORT_NAME parser)
target_include_directories(tsql_parser
    PUBLIC $<BUILD_INTERFACE:${TSQL_PARSER_ROOT}/include> $<INSTALL_INTERFACE:include>
    PRIVATE ${CMAKE_CURRENT_SOURCE_DIR} ${CMAKE_CURRENT_BINARY_DIR})
target_link_libraries(tsql_parser PUBLIC tsql_ast PRIVATE antlr4_static)
target_compile_features(tsql_parser PUBLIC cxx_std_17)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(tsql_parser PRIVATE -Wall -Wextra -Wno-unused-parameter)
endif()

add_executable(tsql_dump tsql_dump.cpp)
target_link_libraries(tsql_dump PRIVATE tsql_parser)

add_executable(tsql_bench ${PROJECT_SOURCE_DIR}/tools/bench/bench.cpp)
target_link_libraries(tsql_bench PRIVATE tsql_parser)
target_include_directories(tsql_bench PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
